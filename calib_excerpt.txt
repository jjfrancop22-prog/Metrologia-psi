function laminarMetricRisk(m){

 if(m.boolean)return m.value?0:1.2;
 if(m.lower!=null&&m.upper!=null){if(m.value<m.lower||m.value>m.upper)return 1.2;const span=m.upper-m.lower;const margin=Math.min(m.value-m.lower,m.upper-m.value);return Math.max(0,1-(margin/(span/2)));}
 if(m.upper!=null){if(m.value>m.upper)return 1.2;return Math.max(0,m.value/m.upper)}
 if(m.lower!=null){if(m.value<m.lower)return 1.2;return Math.max(0,m.lower/Math.max(m.value,1e-9))}
 return null;
}
